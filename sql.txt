CREATE function [dbo].[SalesByDay](@branch int,@InvType int,@StartDate  dateTime,@EndDate  dateTime )  
returns table  
as
return(select InvDate ,sum(SaleTotal - ReturnSaleTotal) as Total,sum(SaleTax - ReturnSaleTax) as TotalTax,  sum(InvTotal-ReturnTotal) as SaleNet,ROW_NUMBER() OVER (ORDER BY InvDate DESC) AS RowNum from
(select  CAST(date AS DATE) as InvDate, sum(tot_net) as InvTotal, sum(inv.InvTotal) as SaleTotal, sum(tax) as SaleTax ,
0 as InvReturnCount,0 as ReturnTotal,0 as  ReturnSaleTotal,0 as ReturnSaleTax
from inv where (@branch is null or  branch =@branch ) and (inv_type=@InvType) and proc_type=1 and IS_Deleted=0 and pay_type <>5 and (@StartDate is null or date>=@StartDate)and (@EndDate is null or date<=@EndDate)  group by date
union all
select  CAST(date AS DATE) as InvDate ,0 as InvCount,0 as InvTotal,0 as SaleTotal,0 as SaleTax, 
sum(tot_net) as ReturnTotal , sum(inv.InvTotal) as ReturnSaleTotal,sum(tax) as ReturnSaleTax
from inv where ( @branch is null or  branch =@branch  ) and (inv_type=@InvType) and proc_type=2 and IS_Deleted=0 and pay_type <>5 and (@StartDate is null or date>=@StartDate)and (@EndDate is null or date<=@EndDate)  group by date  )  as tb group by tb.InvDate )
GO

CREATE FUNCTION [dbo].[ItemsExpirationStock]()
RETURNS TABLE
AS
RETURN(select ItemId,SUM(stockIn-stockOut) as ItemStock,ItemExpire,items.name as ItemName,items.code as ItemCode,Safes.name as StoreName,tb.store as StoreID,DATEDIFF(YEAR, GETDATE(), ItemExpire) as RemaindeYear , DATEDIFF(MONTH, GETDATE(), ItemExpire) as RemaindeMonths,DATEDIFF(DAY, GETDATE(), ItemExpire) as RemaindeDays,items.group_id as GroupID,tb.BranchId from(
   select inv.branch as BranchId,Inv_Sub.store as store, ItemId,sum(val) as stockIn, 0 as stockOut ,CAST(expire_date AS DATE) as ItemExpire from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where  inv_sub.proc_type=1   and inv.IS_Deleted=0 and Items.ItemProperty=8  group by ItemId,inv_sub.store,CAST(expire_date AS DATE) ,inv.branch
union select inv.branch as BranchId,Inv_Sub.store as store, ItemId,0 as stockIn, sum(val) as stockOut,CAST(expire_date AS DATE) as ItemExpire from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=2   and  inv.IS_Deleted=0 and Items.ItemProperty=8 group by ItemId,inv_sub.store,CAST(expire_date AS DATE),inv.branch) as tb 
left join Items on tb.ItemId=items.id left join Safes on Safes.id=tb.store group by ItemExpire,ItemId,items.name,items.code,Safes.name,items.group_id,tb.BranchId,tb.store)

CREATE VIEW [dbo].[vw_Items_Salla_Status] AS
SELECT 
    i.id AS ItemId,
    i.Code AS ItemCode,
    i.name AS ItemName,
    i.nameEN AS ItemNameEn,
    i.barcode AS Barcode,
    i.sale_price AS SalePrice,
    i.purch_price AS PurchPrice,
    ic.name AS CategoryName,
    u.name AS UnitName,
    ISNULL((SELECT SUM(val1) FROM inv_sub WHERE ItemId = i.id), 0) AS CurrentQuantity,
    
    -- بيانات سلة
    ss.SallaProductId,
    ss.IsExported,
    ss.ExportStatus,
    ss.FirstExportDate,
    ss.LastExportDate,
    ss.LastSyncDate,
    ss.ErrorMessage,
    ss.RetryCount,
    -- مقارنة البيانات
    CASE 
        WHEN ss.LastExportedPrice != i.sale_price THEN 1
        WHEN ss.LastExportedCost != i.purch_price THEN 1
        WHEN ss.LastExportedName != i.name THEN 1
        ELSE 0
    END AS HasChanges
    
FROM Items i
LEFT JOIN ItemsCategory ic ON i.group_id = ic.id
LEFT JOIN units u ON i.unit = u.id
LEFT JOIN Items_Salla_Sync ss ON i.id = ss.ItemId
WHERE i.IS_Deleted = 0 ;

CREATE FUNCTION [dbo].[funGetInvoiceItemDetails]
    (@ItemSerialNo NVARCHAR(50) = NULL,
	@Branch INT = NULL,
	@ItemId INT = NULL,
	@StartDate DATE = NULL,
    @EndDate DATE = NULL)
RETURNS TABLE
AS
RETURN
(
    SELECT
        InvoiceItemDetail.ItemId,
        InvoiceItemDetail.ItemSerialNo,
		InvoiceItemDetail.InvertoryImpact,
		InvoiceItemDetail.InvGlobalID,
        inv.inv_type,
		inv.proc_type,
		inv.date as InvDate,
		inv.id as InvNo,
        items.name AS ItemName,
        items.code AS ItemCode,
		Branches.name as BranchName
    FROM
        InvoiceItemDetail
    LEFT JOIN
        Inv ON InvoiceItemDetail.InvGlobalID = inv.InvGlobalID
    LEFT JOIN
        Items ON InvoiceItemDetail.ItemId = items.id
	LEFT JOIN
	   Branches ON inv.branch = Branches.branchID
    WHERE
        (@ItemSerialNo IS NULL OR ItemSerialNo = @ItemSerialNo)
        AND (@Branch IS NULL OR inv.branch = @Branch)
		AND (@ItemId IS NULL OR InvoiceItemDetail.ItemId=@ItemId)
        AND inv.IS_Deleted = 0
		AND inv.proc_type <> 4
		AND (
            (@StartDate IS NULL AND @EndDate IS NULL)
            OR (inv.date BETWEEN  @StartDate AND @EndDate )
        )
	
);
GO

CREATE FUNCTION [dbo].[funCalculateSerialNoSummary](
    @ItemSerialNo NVARCHAR(50) = NULL,
    @itemid INT = NULL,
	@Branch INT = NULL
)
RETURNS TABLE
AS
RETURN
(
    SELECT
        itemid,
        ItemSerialNo,
        SUM(stockIn - stockOut) AS result
    FROM
    (
        SELECT
            itemid,
            ItemSerialNo,
            SUM(ItemQuantity) AS stockIn,
            0 AS stockOut
        FROM
            InvoiceItemDetail
        LEFT JOIN
            Inv ON InvoiceItemDetail.InvGlobalID = Inv.InvGlobalID
        WHERE
			 InvoiceItemDetail.InvertoryImpact=1 
            AND Inv.IS_Deleted = 0
            AND (@ItemSerialNo IS NULL OR ItemSerialNo = @ItemSerialNo)
            AND (@itemid IS NULL OR itemid = @itemid)
        GROUP BY
            itemid,
            ItemSerialNo

        UNION ALL

        SELECT
            itemid,
            ItemSerialNo,
            0 AS stockIn,
            SUM(ItemQuantity) AS stockOut
        FROM
            InvoiceItemDetail
        LEFT JOIN
            Inv ON InvoiceItemDetail.InvGlobalID = Inv.InvGlobalID
        WHERE
            Inv.IS_Deleted = 0
			AND InvoiceItemDetail.InvertoryImpact=2
            AND (@ItemSerialNo IS NULL OR ItemSerialNo = @ItemSerialNo)
            AND (@itemid IS NULL OR itemid = @itemid)
			AND (@Branch IS NULL OR inv.branch = @Branch)
        GROUP BY
            itemid,
            ItemSerialNo
    ) AS tb
    GROUP BY
        itemid,
        ItemSerialNo
);
GO

CREATE PROCEDURE [dbo].[CalcItemStock]
	@branch [int],
	@Inventory [int],
	@item [int],
	@stock Float out
WITH EXECUTE AS CALLER
AS
BEGIN

 set @stock = (select  sum(stockIn - stockOut) as ItemStock from (

	if (@stock is null) set @stock=0
END

Create PROCEDURE [dbo].[GetAccountStatement]
    @ParentCode NVARCHAR(50),
    @StartDate DATETIME,
    @EndDate DATETIME,
    @Branch NVARCHAR(50),
    @ShowPreviousBalance BIT
AS
BEGIN
    SET NOCOUNT ON;

    -- CTE to get account hierarchy
    WITH AccountHierarchy AS (
        SELECT 
            Code,
            AName,
            Type,
            ParentCode
        FROM 
            Accounts_Index
        WHERE 
            Code = @ParentCode
        
        UNION ALL
        
        SELECT 
            a.Code,
            a.AName,
            a.Type,
            a.ParentCode
        FROM 
            Accounts_Index a
        INNER JOIN 
            AccountHierarchy ah ON a.ParentCode = ah.Code
    ),

    -- CTE to calculate previous balance for child accounts
    PreviousBalance AS (
        SELECT 
            @ParentCode AS Code,
             SUM(es.dept) as PreviousDept,
			 SUM(es.credit) AS PreviousCredit
        FROM 
            entry e
        INNER JOIN 
            entry_sub es ON e.GlobalID = es.EntryGlobalID
        WHERE 
            es.acc_no IN (SELECT Code FROM AccountHierarchy WHERE Type = 2)
            AND e.is_Deleted = 0 -- Exclude deleted entries if needed
            AND e.date < @StartDate
            AND (@Branch IS NULL OR e.branch = @Branch)
    ),

    -- CTE to get the main statement
    MainStatement AS (
        SELECT 
            ROW_NUMBER() OVER (ORDER BY e.date) AS DgvRank,
            es.acc_no AS DgvAccCode, 
            ah.AName AS DgvAccName,
            e.doc_No AS DgvEntryNo, 
            es.dept AS DgvDept, 
            es.credit AS DgvCredit,
            e.date AS DgvDate,
            es.notes AS DgvNote,
            es.EntryGlobalID AS DgvGenralNo,
            b.name AS DgvBranch,
			0 as DgvBalance,
            '' DgvBalanceStatus,
            et.Name AS DgvEntryType
        FROM 
            entry e
        INNER JOIN 
            entry_sub es ON e.GlobalID = es.EntryGlobalID
        INNER JOIN 
            AccountHierarchy ah ON es.acc_no = ah.Code
		LEFT JOIN 
            Branches b ON e.branch = b.BranchId
			LEFT JOIN
			EntryTypes et on e.type=et.Id 
        WHERE 
            es.acc_no IN (SELECT Code FROM AccountHierarchy)
            AND e.is_Deleted = 0 -- Exclude deleted entries if needed
            AND (@StartDate IS NULL OR e.date >= @StartDate)
            AND (@EndDate IS NULL OR e.date <= @EndDate)
            AND (@Branch IS NULL OR e.branch = @Branch)
    )

    -- Combine the previous balance with the main statement based on @ShowPreviousBalance
    SELECT
        DgvRank,
        DgvAccCode,
        DgvAccName,
        DgvEntryNo,
        DgvDept,
        DgvCredit,
        DgvDate,
        DgvNote,
        DgvGenralNo,
        DgvBranch,
        DgvBalance,
        DgvBalanceStatus,
        DgvEntryType
    FROM (
        -- Previous balance row
        SELECT
            0 AS DgvRank, -- Use -1 to ensure it appears first
            pb.Code AS DgvAccCode,
            ai.AName AS DgvAccName,
            NULL AS DgvEntryNo,
            pb.PreviousDept AS DgvDept,
            pb.PreviousCredit AS DgvCredit,
            @StartDate AS DgvDate,
            'Previous Balance' AS DgvNote,
            NULL AS DgvGenralNo,
            NULL AS DgvBranch,
            0 AS DgvBalance,
		  'Previous Balance' DgvBalanceStatus,
            'Previous Balance' AS DgvEntryType
        FROM
            PreviousBalance pb
        INNER JOIN
            Accounts_Index ai ON pb.Code = ai.Code
        WHERE @ShowPreviousBalance = 1 

        UNION ALL

        -- Main statement rows
        SELECT
            ms.DgvRank,
            ms.DgvAccCode,
            ms.DgvAccName,
            ms.DgvEntryNo,
            ms.DgvDept,
            ms.DgvCredit,
            ms.DgvDate,
            ms.DgvNote,
            ms.DgvGenralNo,
            ms.DgvBranch,
            ms.DgvBalance,
            ms.DgvBalanceStatus,
            ms.DgvEntryType
        FROM
            MainStatement ms
    ) AS CombinedResult
    ORDER BY
        DgvRank, DgvDate;
END;
GO

WITH EXECUTE AS CALLER
AS
BEGIN

(select  ItemId ,  sum(val1*exchange_price) as Sales ,0 as ReturnSales
 from inv  left join inv_sub on inv.InvGlobalID =Inv_Sub.InvGlobalID   where (@branch is null or branch =@branch ) and (inv_type=@inv_type ) and inv.proc_type=1 and (@StartDate is null or date>=@StartDate)and (@EndDate is null or date<=@EndDate)group by ItemId 
union
 select   ItemId ,  0 as Sales ,sum(val1*exchange_price) as ReturnSales
 from inv  left join inv_sub on inv.InvGlobalID =Inv_Sub.InvGlobalID  where (@branch is null or branch =@branch ) and (inv_type=@inv_type) and inv.proc_type=2 and (@StartDate is null or date>=@StartDate)and (@EndDate is null or date<=@EndDate)group by ItemId   ) as tb left join Items on Items.id= tb.ItemId left join ItemsCategory on Items.group_id= ItemsCategory.CategoryId  group by CategoryID, ItemsCategory.name  order by SaleNet desc
 END

CREATE PROCEDURE [dbo].[InventoryCost]
	@branch [int],
	@date datetime,
	@AvrgCost float out
WITH EXECUTE AS CALLER
AS
BEGIN
	SET NOCOUNT ON;
	set @AvrgCost = (select  sum(ItemAvrgCost * Stock) as AvrgCost from (
select itemId,sum(ItemAvrgCost) as ItemAvrgCost,sum(Stock)as Stock from(
select  ItemId as ItemId,((sum (ItemAvrgCostIn) - sum(ItemAvrgCostOut))+ sum(ItemDiscountIn) -sum(ItemDiscountOut))/NULLIF(sum(stockIn - stockOut), 0) as ItemAvrgCost,0 as Stock from (
select ItemId,sum(val) as stockIn, 0 as stockOut ,0 as ItemDiscountIn,0 as ItemDiscountOut,sum(inv_sub.exchange_price*inv_sub.val1) as ItemAvrgCostIn ,0 as ItemAvrgCostOut from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=1  and inv_sub.store is not null and inv.branch=@branch  and inv.IS_Deleted=0 group by ItemId,inv_sub.store
union select ItemId,0 as stockIn, sum(val)  as stockOut ,0 as ItemDiscountIn,0 as ItemDiscountOut, 0 as ItemAvrgCostIn ,sum(inv_sub.exchange_price*inv_sub.val1) as ItemAvrgCostOut from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=2  and (inv_type=1  or inv_type= 9  or inv_type= 4 or inv_type= 8)  and inv_sub.store is not null and inv.branch=@branch  and inv.IS_Deleted=0 group by ItemId,inv_sub.store
 union select ItemId,0 as stockIn, 0 as stockOut,( sum((((val1*exchange_price)-inv_sub.discount)/(NULLIF(InvTotal, 0)+ minus))* minus)+sum(inv_sub.discount)) as ItemDiscountIn,0 as ItemDiscountOut ,0 as ItemAvrgCostIn,0 as ItemAvrgCostOut  from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=2 and (inv_type=1  or inv_type= 9  or inv_type= 4 or inv_type= 8)   and inv_sub.store is not null and inv.branch=@branch and   inv.IS_Deleted=0 group by ItemId,inv_sub.store
 union select ItemId,0 as stockIn, 0 as stockOut,0 as ItemDiscountIn ,( sum((((val1*exchange_price)-inv_sub.discount)/(NULLIF(InvTotal, 0)+ minus))* minus)+sum(inv_sub.discount)) as ItemDiscountOut,0 as ItemAvrgCostIn,0 as ItemAvrgCostOut  from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=1 and (inv_type=1  or inv_type= 9  or inv_type= 4 or inv_type= 8)   and inv_sub.store is not null and inv.branch=@branch and   inv.IS_Deleted=0 group by ItemId,inv_sub.store
  ) as tb group by ItemId
union select  itemId,0 as ItemAvrgCost, sum(stockIn - stockOut)as Stock from (select ItemId,sum(val) as stockIn, 0 as stockOut   from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=1 and inv_sub.store is not null and inv.branch=@branch   and inv.IS_Deleted=0 group by ItemId,inv_sub.store 
union select ItemId,0 as stockIn, sum(val) as stockOut from inv left join  inv_sub on inv.InvGlobalId=inv_sub.InvGlobalId left join Items on inv_sub.ItemId= Items.id  where inv_sub.proc_type=2  and inv_sub.store is not null and inv.branch=@branch  and  inv.IS_Deleted=0 group by ItemId,inv_sub.store) as tb group by ItemId)as tb group by ItemId)as tb where Stock>0  )
 	if (@AvrgCost is null) set @AvrgCost=0
	if (@AvrgCost <0) set @AvrgCost=0
END

CREATE PROCEDURE [dbo].[MonthlySales]
	@branch  [int] ,
	@inv_type  [int] ,
	@pay_type  [int] ,
	@Seller  [int] 
WITH EXECUTE AS CALLER
AS
BEGIN
select InvDate,InvYear,InvMonth , sum(InvCount) as InvCount,sum(InvReturnCount) as InvReturnCount ,sum(InvTotal)as InvTotal , sum(ReturnTotal) as ReturnTotal, sum(InvTotal-ReturnTotal) as SaleNet from

CREATE PROCEDURE [dbo].[proGetCategorySaleByDay]
    @branch INT,
    @inv_type INT,
    @StartDate DATE,
    @EndDate DATE,
	@CategoryID INT
AS
BEGIN
    SELECT 
        ItemsCategory.name AS CategoryName,
        ItemsCategory.Code AS CtgyCode,
        ItemsCategory.CategoryID,
        SUM(Sales) AS Sales,
        SUM(ReturnSales) AS ReturnSales,
        SUM(Sales - ReturnSales) - SUM(InvoiceDiscount1 - InvoiceDiscount2) AS SaleNet,
        InvDate
    FROM
    (
        SELECT
            ItemId ,  
			SUM((val1 * exchange_price) - discount) as Sales 
			,0 as ReturnSales,
			SUM(round((exchange_price * minus  /InvSum),2)) InvoiceDiscount1,
			0 InvoiceDiscount2,
			CAST(inv.date AS DATE) as InvDate
        FROM Inv_Sub
        LEFT JOIN inv ON Inv_Sub.InvGlobalID = Inv.InvGlobalID
        WHERE
             Inv.IS_Deleted = 0
			AND Inv_Sub.ProductId=0
            AND (@branch IS NULL OR branch = @branch)
            AND inv_type = @inv_type
			AND  inv.proc_type=1
            AND (@StartDate IS NULL OR date >= @StartDate)
            AND (@EndDate IS NULL OR date <= @EndDate)
        GROUP BY ItemId, inv.date
        UNION ALL
        SELECT
           ItemId , 
		   0 as Sales ,
		   SUM((val1 * exchange_price) - discount) as ReturnSales,
		   0 InvoiceDiscount1,
		   SUM(round((exchange_price * minus  /InvSum),2)) InvoiceDiscount2,
		   CAST(inv.date AS DATE) as InvDate
         FROM Inv_Sub
        LEFT JOIN inv ON Inv_Sub.InvGlobalID = Inv.InvGlobalID
        WHERE
             Inv.IS_Deleted = 0
			 AND Inv_Sub.ProductId=0
            AND (@branch IS NULL OR branch = @branch)
            AND inv_type = @inv_type
			AND  inv.proc_type=2
            AND (@StartDate IS NULL OR date >= @StartDate)
            AND (@EndDate IS NULL OR date <= @EndDate)
        GROUP BY ItemId, inv.date
    ) AS tb
	
    LEFT JOIN Items ON Items.id = tb.ItemId
    LEFT JOIN ItemsCategory ON Items.group_id = ItemsCategory.CategoryId
	where  (@CategoryID IS NULL OR ItemsCategory.CategoryID = @CategoryID)
    GROUP BY CategoryID, ItemsCategory.name, ItemsCategory.Code, tb.InvDate
    ORDER BY InvDate DESC;
END
